            if (( at[$i] <= current_time && finished[$i] == 0 )); then

            fi
        done
        
        if (( ${#ready_pids[@]} == 0 )); then
            current_time=$((current_time + 1))
            continue
        fi
        
        # Find process with shortest burst time
        min_bt=9999
        selected_pid=-1
        for pid in "${ready_pids[@]}"; do
            if (( bt[$pid] < min_bt )); then
                min_bt=${bt[$pid]}
                selected_pid=$pid
            fi
        done
        
        # Execute process
        current_time=$((current_time + bt[$selected_pid]))
        ct[$selected_pid]=$current_time
        finished[$selected_pid]=1
        completed_processes=$((completed_processes + 1))
    done

        ready_pids=()
        for (( i=1; i<=n; i++ )); do

                ready_pids+=($i)
            fi
        done
        
        if (( ${#ready_pids[@]} == 0 )); then
            current_time=$((current_time + 1))
            continue
        fi
        
        # Find process with shortest *remaining* burst time
        min_rbt=9999
        selected_pid=-1
        for pid in "${ready_pids[@]}"; do
            if (( rbt[$pid] < min_rbt )); then
                min_rbt=${rbt[$pid]}
                selected_pid=$pid
            fi
        done
        
        # Execute process for one time unit

        current_time=$((current_time + 1))
        
        if (( rbt[$selected_pid] == 0 )); then

def priority_scheduling(processes, choice, preemptive):
    n = len(processes)
    
    # Store original burst time and priority for lookup
    burst_time = {p[3]: p[1] for p in processes}
    priority = {p[3]: p[2] for p in processes}
    arrival_time = {p[3]: p[0] for p in processes}
    
    # Use remaining_bt indexed by original pid
    remaining_bt = burst_time.copy()
    
    wt = {p[3]: 0 for p in processes}
    tat = {p[3]: 0 for p in processes}
    ct = {p[3]: 0 for p in processes}
    
    t = 0
    completed = 0

    def better_priority(p1_pri, p2_pri):
        if choice == "high":
            return p1_pri > p2_pri
        else: # low
            return p1_pri < p2_pri

    if preemptive:
        while completed != n:
            idx = -1
            best_pri = None
            
            # Find best priority process in ready queue
            for p in processes:
                pid = p[3] # Original PID
                if remaining_bt[pid] > 0 and arrival_time[pid] <= t:
                    if idx == -1 or better_priority(priority[pid], best_pri):
                        idx = pid
                        best_pri = priority[pid]
            
            if idx != -1:
                remaining_bt[idx] -= 1
                
                if remaining_bt[idx] == 0:
                    ct[idx] = t + 1
                    completed += 1
            # else: CPU is idle
            
            t += 1
            
    else: # Non-preemptive
        while completed != n:
            ready_processes = []
            for p in processes:
                pid = p[3]
                if remaining_bt[pid] > 0 and arrival_time[pid] <= t:
                    ready_processes.append(pid)
            
            if not ready_processes:
                t += 1
                continue

            selected_idx = -1
            best_pri = None
            
            # Find best priority in the ready queue
            for pid in ready_processes:
                if selected_idx == -1 or better_priority(priority[pid], best_pri):
                    selected_idx = pid
                    best_pri = priority[pid]
            
            # Run the selected process to completion
            t += remaining_bt[selected_idx]
            ct[selected_idx] = t
            remaining_bt[selected_idx] = 0
            completed += 1

    # Calculate TAT and WT
    for p in processes:
        pid = p[3]
        tat[pid] = ct[pid] - arrival_time[pid]
        wt[pid] = tat[pid] - burst_time[pid]
    
    return ct, tat, wt

declare -A at bt pri rbt ct tat wt finished

read -p "Enter number of processes: " n
read -p "Priority scheme (high/low): " pri_choice
pri_choice=$(echo "$pri_choice" | tr 'A-Z' 'a-z')
read -p "Preemptive? (y/n): " preemptive_choice
preemptive_choice=$(echo "$preemptive_choice" | tr 'A-Z' 'a-z')

for (( i=1; i<=n; i++ )); do
    read -p "Enter Arrival Time for P$i: " at[$i]
    read -p "Enter Burst Time for P$i: " bt[$i]
    read -p "Enter Priority for P$i: " pri[$i]
    rbt[$i]=${bt[$i]}
    finished[$i]=0
done

current_time=0
completed_processes=0
total_tat=0
total_wt=0

# Helper function to compare priorities
# Usage: better_priority $pri1 $pri2
# Returns 0 (true) if p1 is better, 1 (false) if p2 is better
better_priority() {
    local p1_pri=$1
    local p2_pri=$2
    if [[ "$pri_choice" == "high" ]]; then
        (( p1_pri > p2_pri )) && return 0 || return 1
    else
        (( p1_pri < p2_pri )) && return 0 || return 1
    fi
}

if [[ "$preemptive_choice" == "y" ]]; then
    # Preemptive
    while (( completed_processes < n )); do
        ready_pids=()
        for (( i=1; i<=n; i++ )); do
            if (( at[$i] <= current_time && finished[$i] == 0 )); then
                ready_pids+=($i)
            fi
        done
        
        if (( ${#ready_pids[@]} == 0 )); then
            current_time=$((current_time + 1))
            continue
        fi
        
        selected_pid=-1
        best_pri=-1
        if [[ "$pri_choice" == "low" ]]; then best_pri=9999; fi

        for pid in "${ready_pids[@]}"; do
            if [[ $selected_pid -eq -1 ]] || ( better_priority ${pri[$pid]} $best_pri ); then
                selected_pid=$pid
                best_pri=${pri[$pid]}
            fi
        done
        
        rbt[$selected_pid]=$((rbt[$selected_pid] - 1))
        current_time=$((current_time + 1))
        
        if (( rbt[$selected_pid] == 0 )); then
            ct[$selected_pid]=$current_time
            finished[$selected_pid]=1
            completed_processes=$((completed_processes + 1))
        fi
    done
else
    # Non-Preemptive
    while (( completed_processes < n )); do
        ready_pids=()
        for (( i=1; i<=n; i++ )); do
            if (( at[$i] <= current_time && finished[$i] == 0 )); then
                ready_pids+=($i)
            fi
        done
        
        if (( ${#ready_pids[@]} == 0 )); then
            current_time=$((current_time + 1))
            continue
        fi
        
        selected_pid=-1
        best_pri=-1
        if [[ "$pri_choice" == "low" ]]; then best_pri=9999; fi

        for pid in "${ready_pids[@]}"; do
            if [[ $selected_pid -eq -1 ]] || ( better_priority ${pri[$pid]} $best_pri ); then
                selected_pid=$pid
                best_pri=${pri[$pid]}
            fi
        done
        
        current_time=$((current_time + rbt[$selected_pid]))
        rbt[$selected_pid]=0
        ct[$selected_pid]=$current_time
        finished[$selected_pid]=1
        completed_processes=$((completed_processes + 1))
    done
fi

# --- Final Calculations & Printing ---
echo
printf "%-5s | %-5s | %-5s | %-5s | %-5s | %-5s | %-5s\n" "PID" "AT" "BT" "Pri" "CT" "TAT" "WT"
echo "---------------------------------------------------"
for (( i=1; i<=n; i++ )); do
    tat[$i]=$((ct[$i] - at[$i]))
    wt[$i]=$((tat[$i] - bt[$i]))
    
    total_tat=$((total_tat + tat[$i]))
    total_wt=$((total_wt + wt[$i]))
    
    printf "P%-4s | %-5s | %-5s | %-5s | %-5s | %-5s | %-5s\n" "$i" "${at[$i]}" "${bt[$i]}" "${pri[$i]}" "${ct[$i]}" "${tat[$i]}" "${wt[$i]}"
done

avg_tat=$(echo "scale=2; $total_tat / $n" | bc)
avg_wt=$(echo "scale=2; $total_wt / $n" | bc)

echo -e "\nAverage Turnaround Time: $avg_tat"
echo "Average Waiting Time: $avg_wt"

def enter_cs(p, other, timeout=2.0):
    global flag, turn
    
    # Step 1: Raise flag
    flag[p] = True
    # Step 2: Set turn
    turn = other
    
    start = time.time()
    
    # Step 3: Busy-wait
    while flag[other] and turn == other:
        # Check for timeout (for this simulation)
        if time.time() - start > timeout:
            print(f"P{p} timed out waiting!")
            return False
        
        # Politely wait
        time.sleep(0.01)
        
    # If loop exits, it's safe to enter
    return True

def exit_cs(p):
    global flag
    # Step 4: Lower flag
    flag[p] = False

# These are blocking, "busy-wait" semaphores
def wait(s):
    while s <= 0:
        pass # Wait
    return s - 1

    return s + 1

def producer():
    global mutex, empty, full, buf
    
    global empty
    empty = wait(empty) # Wait for an empty slot
    
    global mutex
    mutex = wait(mutex) # Lock the buffer
    
    # Critical Section
    print("Producer is producing...")
    buf[full] = 1 # 'full' acts as the 'in' index
    print("Producer has produced one item.")
    # End Critical Section
    
    mutex = signal(mutex) # Unlock the buffer
    
    global full
    full = signal(full) # Signal a full slot

def consumer():
    global mutex, empty, full, buf
    
    global full
    full = wait(full) # Wait for a full slot
    
    global mutex
    mutex = wait(mutex) # Lock the buffer
    
    # Critical Section
    print("Consumer is consuming...")
    buf[full] = 0 # 'full' is 0-indexed, so full=1 means index 0
    print("Consumer has consumed one item.")
    # End Critical Section
    
    mutex = signal(mutex) # Unlock the buffer
    
    global empty
    empty = signal(empty) # Signal an empty slot

# Non-blocking wait(semaphore_name)
# Returns 0 for success, 1 for failure
wait() {
    local s_name=$1
    eval "s=\$$s_name"
    
    if [[ $s -le 0 ]]; then
        return 1 # Failure
    fi
    
    s=$((s - 1))
    eval "$s_name=$s"
    return 0 # Success
}

# signal(semaphore_name)
signal() {
    local s_name=$1
    eval "s=\$$s_name"
    s=$((s + 1))
    eval "$s_name=$s"
}

writer() {
    if wait wrt; then
        # --- Critical Section ---
        echo "Writer is writing..."
        sleep 1
        # --- End Critical Section ---
        signal wrt
        echo "Writer has finished writing..."
    else
        echo "-> Writer can't enter now, resource busy."
    fi
}

reader() {
    if ! wait mutex; then
        echo "-> Reader waiting (can't access readcount)."
        return
    fi
    
    readcount=$((readcount + 1))
    if [[ $readcount -eq 1 ]]; then
        if ! wait wrt; then # Try to lock out writers
            echo "-> Reader waiting (writer active)."
            readcount=$((readcount - 1)) # Rollback
            signal mutex
            return
        fi
    fi
    
    signal mutex
    
    # --- Critical Section ---
    echo "Reader started reading. Total readers: $readcount"
    
    echo "--- Reader is active ---"
    echo "1 -> Add another reader"
    echo "2 -> Add a writer (will fail)"
    echo "3 -> Stop reading (this reader finishes)"
    
    while true; do
        read -p "Reader choice: " rchoice
        case $rchoice in
            1) 
               # Simplified simulation: just increment count
               wait mutex
               readcount=$((readcount + 1))
               echo "Another reader started. Total readers: $readcount"
               signal mutex
               ;;
            2) writer ;; # Simulate a writer (will fail)
            3) break ;;
            *) echo "Invalid choice (1, 2, or 3)" ;;
        esac
    done
    
    # --- End Critical Section ---
    
    # --- Exit Protocol ---
    wait mutex # We know this will succeed
    readcount=$((readcount - 1))
    echo "Reader stops reading. Remaining readers: $readcount"
    if [[ $readcount -eq 0 ]]; then
        signal wrt # Last reader out
    fi
    signal mutex
}

# --- Vector Math Helper Functions ---
# leq(a, b): Returns 1 if (a <= b) element-wise, 0 otherwise
leq() {
  local a=($1) b=($2) i
  for ((i=0;i<${#a[@]};i++)); do
    if (( a[i] > b[i] )); then echo 0; return; fi
  done
  echo 1
}

# add(a, b): Returns vector a + b
add() {
  local a=($1) b=($2) out=() i
  for ((i=0;i<${#a[@]};i++)); do out+=($(( a[i] + b[i] ))); done
  echo "${out[@]}"
}

# sub(a, b): Returns vector a - b
sub() {
  local a=($1) b=($2) out=() i
  for ((i=0;i<${#a[@]};i++)); do out+=($(( a[i] - b[i] ))); done
  echo "${out[@]}"
}

# --- Safety Algorithm ---
# Populates global SAFE, SEQ, and avail_after_when_executed
safety_sequence_rolling() {
  local available_str="$1"
  shift
  local -n allocation_ref=$1
  shift
  local -n need_ref=$1
  local n="${#allocation_ref[@]}"
  local work="$available_str"
  local -a finished
  local -a seq
  local k=0
  declare -gA avail_after_when_executed # Stores results
  local count=0

  while (( count < n )); do
    local found=0
    local jump
    for ((jump=0;jump=1 && pi_1<=n )); then break; fi
    echo "Invalid index."
  done
  pi=$((pi_1-1))
  req="$(read_vec "Enter Request vector for P$pi_1 ($m ints): " "$m")"

  need_pi="${need[$pi]}"
  ok1=$(leq "$req" "$need_pi")
  if [[ $ok1 -ne 1 ]]; then
    echo
    echo "DENIED: Request exceeds Need of P$pi_1."
    exit 0
  fi
  ok2=$(leq "$req" "$available")
  if [[ $ok2 -ne 1 ]]; then
    echo
    echo "WAIT: Request exceeds current Available. Not enough resources now."
    exit 0
  fi

  available_new="$(sub "$available" "$req")"
  allocation_new=("${allocation[@]}")
  need_new=("${need[@]}")
  allocation_new[$pi]="$(add "${allocation_new[$pi]}" "$req")"
  need_new[$pi]="$(sub "${need_new[$pi]}" "$req")"

  safety_sequence_rolling "$available_new" allocation_new need_new
  print_table_P1_to_Pn allocation_new maxm need_new "Table After Tentative Grant (P1..Pn order)"

  if [[ $SAFE -eq 1 ]]; then
    seq_str=""
    for idx in "${SEQ[@]}"; do
      p="P$((idx+1))"
      if [[ -z "$seq_str" ]]; then seq_str="$p"; else seq_str="$seq_str → $p"; fi
    done
    echo
    echo "Request CAN be granted. System remains SAFE."
    echo "Safe Sequence after granting = $seq_str"
  else
    echo
    echo "Request would lead to an UNSAFE state. DENIED."
  fi
}

def leq(a, b): return all(x <= y for x, y in zip(a, b)) 
def add(a, b): return [x+y for x, y in zip(a, b)] 
def sub(a, b): return [x-y for x, y in zip(a, b)] 
 
def safety_sequence_rolling(available, allocation, need): 
    n = len(allocation) 
    work = available[:] 
    finished = [False]*n 
    seq = [] 
    k = 0 
    avail_after_when_executed = {} 
    while len(seq) < n: 
        found = False 
        for jump in range(n): 
            i = (k + jump) % n 
 
            if not finished[i] and leq(need[i], work): 
                new_work = add(work, allocation[i]) 
                finished[i] = True 
                seq.append(i) 
                avail_after_when_executed[i] = new_work[:]   
                work = new_work 
                k = (i + 1) % n 
                found = True 
                break 
        if not found: 
            return False, seq, avail_after_when_executed 
    return True, seq, avail_after_when_executed 
 
def print_table_P1_to_Pn(allocation, maxm, need, avail_after_when_executed, title): 
    n = len(allocation) 
    print(f"\n--- {title} ---") 
    print(f"{'Process':<8} | {'Allocation':<12} | {'Max Need':<12} | {'Current Availability':<22} | {'Remaining Need'}") 
    print("-"*95) 
    for i in range(n): 
        avail_str = str(avail_after_when_executed.get(i, "-")) 
        print(f"P{i+1:<7} | {str(allocation[i]):<12} | {str(maxm[i]):<12} | {avail_str:<22} | {str(need[i])}") 
 
def read_vec(prompt, m): 
    while True: 
        try: 
            v = list(map(int, input(prompt).split())) 
            if len(v) != m: print(f"Enter exactly {m} integers."); continue 
            return v 
        except ValueError: 
            print("Enter integers separated by spaces.") 
 
def read_matrix(prompt_row, n, m): 
    mat = [] 
    for i in range(n): 
        mat.append(read_vec(f"{prompt_row} P{i+1}: ", m)) 
    return mat 
 
def main(): 
    print("=== Banker's Algorithm (P1..Pn Table + Request Handling) ===") 
    n, m = map(int, input("Enter n (processes) and m (resource types): ").split()) 
 
    available = read_vec(f"Enter Available vector ({m} ints): ", m) 
    allocation = read_matrix("Enter Allocation for", n, m) 
    maxm = read_matrix("Enter Max Need for", n, m) 
    need = [sub(maxm[i], allocation[i]) for i in range(n)] 
 
    safe, seq, avail_after = safety_sequence_rolling(available, allocation, need) 
    if not safe: 
        print_table_P1_to_Pn(allocation, maxm, need, avail_after, "Table (Initial, P1..Pn order)") 
        print("\nSystem is NOT in a SAFE state initially. Exiting.") 
        return 
 
    print_table_P1_to_Pn(allocation, maxm, need, avail_after, "Table (Initial, P1..Pn order)") 
    print("\nInitial Safe Sequence = " + " → ".join(f"P{i+1}" for i in seq)) 
 
    opt = input("\nUser Input 5: Any Request? (Yes/No): ").strip().lower() 
    if opt not in ("yes", "y"): 
        print("\nNo request processed. Done.") 
        return 
 
    while True: 
        try: 
            pi_1 = int(input(f"User Input 6: Enter requesting process index Pi (1..{n}): ")) 
 
            if 1 <= pi_1 <= n: break 
            print("Invalid index.") 
        except ValueError: 
            print("Enter an integer.") 
    pi = pi_1 - 1 
    req = read_vec(f"Enter Request vector for P{pi_1} ({m} ints): ", m) 
 
    if not leq(req, need[pi]): 
        print(f"\nDENIED: Request exceeds Need of P{pi_1}.") 
        return 
    if not leq(req, available): 
        print(f"\nWAIT: Request exceeds current Available. Not enough resources now.") 
        return 
 
    available_new = sub(available, req) 
    allocation_new = [row[:] for row in allocation] 
    need_new = [row[:] for row in need] 
    allocation_new[pi] = add(allocation_new[pi], req) 
    need_new[pi] = sub(need_new[pi], req) 
 
    safe2, seq2, avail_after2 = safety_sequence_rolling(available_new, allocation_new, need_new) 
    print_table_P1_to_Pn(allocation_new, maxm, need_new, avail_after2, "Table After Tentative Grant (P1..Pn order)") 
 
    if safe2: 
        print("\nRequest CAN be granted. System remains SAFE.") 
        print("Safe Sequence after granting = " + " → ".join(f"P{i+1}" for i in seq2)) 
    else: 
        print("\nRequest would lead to an UNSAFE state. DENIED.") 
 
if __name__ == "__main__": 
    main()
